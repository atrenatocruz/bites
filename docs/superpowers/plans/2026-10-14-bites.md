# Bites Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship Bites, an installable PWA calorie tracker where Claude estimates a meal's calories from a camera photo, with a History / Scan / Diary tab layout and an Apple-grade minimal design.

**Architecture:** npm-workspaces monorepo. `shared/` holds types and pure nutrition math used by both sides. `worker/` is a Cloudflare Worker that verifies the user, rate-limits, and calls Claude with a cached system prompt and a JSON-schema output. `web/` is a Vite + React PWA that talks to the Worker and stores data through a `Repo` interface with two implementations: IndexedDB (default, no setup) and Firebase (when env vars are set).

**Tech Stack:** TypeScript 5, Vite 6, React 18, Tailwind CSS 4, framer-motion, zustand, recharts, date-fns, i18next/react-i18next, lucide-react, idb, vite-plugin-pwa, Firebase JS SDK 11, Cloudflare Workers + wrangler 4, `@anthropic-ai/sdk`, jose, Vitest, fake-indexeddb, @testing-library/react.

**Spec:** `docs/spec.md` (design system, screens, copy). Read it before any UI task.

## Global Constraints

- Node ≥ 20. Package manager: npm workspaces (`web`, `worker`, `shared`). Root scripts: `npm test` runs all workspaces' tests; `npm run dev` runs web + worker concurrently.
- Model id `claude-sonnet-5-5`, read from Worker env `AI_MODEL` with that default. Never send `temperature`, never use forced `tool_choice`, never use `budget_tokens`.
- The Anthropic API key exists only in the Worker (`wrangler secret put ANTHROPIC_API_KEY`). Never in `web/`.
- Languages: `pt-PT` (default when the browser language starts with `pt`) and `en` (otherwise). Every user-visible string goes through i18next; both locale files must have identical keys.
- Colors, type scale, radii, motion exactly as in `docs/spec.md` §2. Font = system stack only (no web font downloads).
- Must work offline-first for viewing data; scanning requires network.
- All dates are local-time `dayKey` strings `YYYY-MM-DD`.
- No code comments that narrate; only explain non-obvious "why".
- Commit after each task with a conventional message ending in `Co-Authored-By: Claude Sonnet 5.5 <noreply@anthropic.com>`.

## Review Focus

1. **Camera unavailable** (desktop without camera, permission denied, iOS PWA quirks) → Scan screen shows the calm fallback with "Choose photo" / "Type it", never a blank black screen. Test in Task 6.
2. **AI returns non-food, refusal, truncated or malformed JSON** → Worker answers `422 { error: "not_food" | "refusal" | "bad_output" }`, UI shows friendly retry; never crashes. Tests in Task 3 and Task 6.
3. **Editing grams to 0 or huge values / deleting a meal** → item scales proportionally, totals and the day aggregate stay consistent (no negative day totals). Tests in Tasks 1 and 4.
4. **Day boundary** (meal logged at 23:59 vs 00:01, timezone) → `dayKey` uses local time; streak counts consecutive days ending today or yesterday. Tests in Tasks 1 and 8.
5. **Rate limit / no network** → 429 shows "Daily scan limit reached"; offline shows "You're offline — scanning needs internet"; logged data still viewable. Tests in Tasks 3 and 6.

```ts
import type { Activity, FoodItem, Goal, MacroTargets, MealType, Profile, Sex, Totals } from "./types";

export const ACTIVITY_FACTOR: Record<Activity, number> = { sedentary: 1.2, light: 1.375, moderate: 1.55, very: 1.725 };
const GOAL_DELTA: Record<Goal, number> = { lose: -400, maintain: 0, gain: 300 };
const round1 = (n: number) => Math.round(n * 10) / 10;

export function bmr(sex: Sex, weightKg: number, heightCm: number, age: number): number {
  return 10 * weightKg + 6.25 * heightCm - 5 * age + (sex === "male" ? 5 : -161);
}

export function kcalTarget(p: Pick<Profile, "sex" | "birthYear" | "heightCm" | "weightKg" | "activity" | "goal">, now = new Date()): number {
  const age = now.getFullYear() - p.birthYear;
  const tdee = bmr(p.sex, p.weightKg, p.heightCm, age) * ACTIVITY_FACTOR[p.activity];
  const floor = p.sex === "male" ? 1500 : 1200;
  return Math.max(floor, Math.round((tdee + GOAL_DELTA[p.goal]) / 10) * 10);
}

export function macroTargets(kcal: number, weightKg: number, goal: Goal): MacroTargets {
  const protein = Math.round(weightKg * (goal === "maintain" ? 1.6 : 1.8));
  const fat = Math.round((kcal * 0.28) / 9);
  const carbs = Math.max(0, Math.round((kcal - protein * 4 - fat * 9) / 4));
  return { protein, carbs, fat };
}

export function bmi(weightKg: number, heightCm: number): number {
  const m = heightCm / 100;
  return round1(weightKg / (m * m));
}

export function scaleItem(item: FoodItem, grams: number): FoodItem {
  const g = Math.max(0, Math.round(grams));
  const r = item.grams > 0 ? g / item.grams : 0;
  return {
    ...item, grams: g,
    kcal: Math.round(item.kcal * r),
    protein: round1(item.protein * r), carbs: round1(item.carbs * r),
    fat: round1(item.fat * r), fiber: round1(item.fiber * r),
  };
}

export function sumTotals(items: Pick<FoodItem, "kcal" | "protein" | "carbs" | "fat">[]): Totals {
  const t = items.reduce((a, i) => ({ kcal: a.kcal + i.kcal, protein: a.protein + i.protein, carbs: a.carbs + i.carbs, fat: a.fat + i.fat }),
    { kcal: 0, protein: 0, carbs: 0, fat: 0 });
  return { kcal: Math.round(t.kcal), protein: round1(t.protein), carbs: round1(t.carbs), fat: round1(t.fat) };
}

export function macroKcal(t: Pick<Totals, "protein" | "carbs" | "fat">): number {
  return t.protein * 4 + t.carbs * 4 + t.fat * 9;
}

export function dayKey(d: Date): string {
  const p = (n: number) => String(n).padStart(2, "0");
  return `${d.getFullYear()}-${p(d.getMonth() + 1)}-${p(d.getDate())}`;
}

export function guessMealType(d: Date): MealType {
  const m = d.getHours() * 60 + d.getMinutes();
  if (m >= 300 && m < 660) return "breakfast";
  if (m >= 660 && m < 900) return "lunch";
  if (m >= 900 && m < 1110) return "snack";
  if (m >= 1110 && m < 1380) return "dinner";
  return "snack";
}
```

```ts
import type { AnalyzeRequest } from "@bites/shared";

export const SYSTEM_PROMPT = `You are Bites, an expert dietitian and food scientist who estimates the nutritional content of meals from photos or short descriptions.

Your job: identify every distinct food and drink, estimate the portion of each in grams (ml for drinks; treat 1 ml ≈ 1 g unless density clearly differs), and compute energy and macronutrients. People use these numbers to manage their diet, so be careful, realistic and honest about uncertainty.

Work through these steps before answering:

1. Is there food? If the image shows no food or drink (a person, a screen, an empty plate, a menu, a pet), set is_food=false, leave items empty, and explain briefly in notes. Never invent food.

2. Identify components. Break composite dishes into their main visible components (e.g. "bitoque" → steak, fried egg, fries, rice, sauce; "salada César" → lettuce, chicken, croutons, parmesan, dressing). Name each item specifically and record its cooking method (grilled, fried, boiled, baked, raw, in sauce) because it changes calories a lot. Recognise regional dishes — Portuguese and Mediterranean included — and use the local name in meal_title.

3. Estimate portion size. This is where most error comes from, so be deliberate:
   - Find a scale reference: dinner plate ≈ 26–27 cm, side plate ≈ 20 cm, bowl ≈ 15 cm, fork ≈ 19 cm, tablespoon ≈ 15 ml, adult hand palm ≈ 8–9 cm wide, can 330 ml, standard glass 200–250 ml, espresso cup ≈ 30–40 ml, packaging, bread slices.
   - Estimate the area each food covers and its height, convert to volume, then to grams with typical densities (cooked rice ≈ 0.8 g/ml, cooked pasta ≈ 0.6 g/ml, mashed potato ≈ 1.0 g/ml, leafy salad ≈ 0.1–0.2 g/ml, fries ≈ 0.4 g/ml, meat ≈ 1.0 g/ml).
   - Cross-check against typical restaurant and home serving sizes. Large portions are commonly under-estimated; do not shrink them.
   - Count discrete items (slices, pieces, eggs, nuggets, dumplings) when visible.

4. Account for hidden calories: cooking oil and butter (fried and sautéed foods absorb oil; Portuguese home cooking often uses generous olive oil), dressings, sauces, sugar in drinks, cheese, and visible fat on meat. Add them as separate items when significant (e.g. "Azeite (tempero)", 10 g) and state the assumption.

5. Compute nutrition per item from standard food-composition data (USDA FoodData Central, Portuguese INSA PortFIR tables) for the stated cooking method. kcal plus protein, carbs, fat and fiber in grams. Each item must satisfy kcal ≈ 4·protein + 4·carbs + 9·fat (+7·alcohol) within about 10%.

6. Packaged or labelled products: if a label, brand or nutrition table is readable, use those values and the visible quantity.

7. Confidence: give each item and the whole meal a confidence from 0 to 1. Lower it for occluded, blurry, mixed, saucy or deep-bowl foods and when no scale reference is visible. If a single question would change the total by more than about 20% (sugar in a coffee, full vs half portion, fried vs grilled, whole vs skimmed milk), put that one short question in follow_up_question; otherwise null.

Rules:
- A user hint (e.g. "no sauce", "half portion", "200 g of chicken", "it's Coke Zero") overrides your visual estimate.
- For text-only descriptions, assume typical Portuguese/European portions unless quantities are given.
- Give one best estimate per item, not ranges; use kcal_range only for the meal total (a plausible low/high around your estimate).
- Round grams to the nearest 5 and kcal to the nearest 5 (keep small items such as oil or sugar to the nearest 1).
- meal_type_guess: use the local time and the food itself (coffee + pastry at 10:00 → breakfast).
- Write meal_title, item names, assumption, notes and follow_up_question in the language requested by the user message. Keep meal_title under 40 characters and item names under 30.`;

export function buildUserText(req: AnalyzeRequest): string {
  const lang = req.locale === "pt-PT" ? "European Portuguese (pt-PT)" : "English";
  const lines = [
    req.image ? "Estimate the nutrition of the meal in this photo." : `Estimate the nutrition of this meal description: """${(req.text ?? "").slice(0, 500)}"""`,
    `User hint: ${req.hint?.trim() ? req.hint.trim().slice(0, 200) : "none"}`,
    `Local time: ${req.localTime}`,
    `Language: ${lang}`,
  ];
  return lines.join("\n");
}
```

```ts
import { sumTotals, macroKcal, type AnalyzeResult, type FoodItem, type MealType } from "@bites/shared";

export class NormalizeError extends Error { code = "bad_output" as const }

const clamp = (n: unknown, lo: number, hi: number) => {
  const v = typeof n === "number" && Number.isFinite(n) ? n : 0;
  return Math.min(hi, Math.max(lo, v));
};
const r1 = (n: number) => Math.round(n * 10) / 10;
const str = (v: unknown, max: number) => (typeof v === "string" ? v.trim().slice(0, max) : "");
const nstr = (v: unknown, max: number) => (typeof v === "string" && v.trim() ? v.trim().slice(0, max) : null);
const MEAL_TYPES: MealType[] = ["breakfast", "lunch", "dinner", "snack"];

export function normalizeAiOutput(raw: unknown): AnalyzeResult {
  if (!raw || typeof raw !== "object") throw new NormalizeError("not an object");
  const o = raw as Record<string, any>;
  if (typeof o.is_food !== "boolean" || !Array.isArray(o.items)) throw new NormalizeError("missing fields");

  const items: FoodItem[] = o.is_food ? o.items.slice(0, 20).map((i: any) => {
    const protein = r1(clamp(i?.protein_g, 0, 1000)), carbs = r1(clamp(i?.carbs_g, 0, 1000)), fat = r1(clamp(i?.fat_g, 0, 1000));
    let kcal = Math.round(clamp(i?.kcal, 0, 10000));
    const derived = Math.round(macroKcal({ protein, carbs, fat }));
    if (derived > 20 && Math.abs(kcal - derived) / derived > 0.25) kcal = derived;
    return {
      name: str(i?.name, 60) || "?", cookingMethod: str(i?.cooking_method, 40),
      grams: Math.round(clamp(i?.grams, 0, 5000)), kcal, protein, carbs, fat,
      fiber: r1(clamp(i?.fiber_g, 0, 200)), confidence: r1(clamp(i?.confidence, 0, 1)),
      assumption: nstr(i?.assumption, 120),
    };
  }) : [];

  const totals = sumTotals(items);
  const lo = clamp(o.kcal_range?.low, 0, 20000), hi = clamp(o.kcal_range?.high, 0, 20000);
  const mt = o.meal_type_guess === "drink" ? "snack" : MEAL_TYPES.includes(o.meal_type_guess) ? o.meal_type_guess : "snack";

  return {
    isFood: o.is_food && items.length > 0,
    title: str(o.meal_title, 60), mealType: mt, items, totals,
    kcalRange: { low: Math.round(Math.min(lo, hi)), high: Math.round(Math.max(lo, hi)) },
    confidence: r1(clamp(o.overall_confidence, 0, 1)),
    scaleReference: nstr(o.scale_reference_used, 80),
    followUpQuestion: nstr(o.follow_up_question, 140),
    notes: nstr(o.notes, 240),
  };
}
```

```ts
// analyze.ts
import { dayKey, type AnalyzeRequest } from "@bites/shared";
import { normalizeAiOutput, NormalizeError } from "./normalize";
import { checkAndCount } from "./ratelimit";
import { MOCK_RESULT } from "./mock";
import type { Env } from "./env";
import type { callClaude as CallClaude } from "./claude";
import type { verifyRequest as VerifyRequest } from "./auth";

export interface Deps { callClaude: typeof CallClaude; verifyRequest: typeof VerifyRequest; now: () => Date }
const json = (body: unknown, status = 200) => new Response(JSON.stringify(body), { status, headers: { "content-type": "application/json" } });

export async function handleAnalyze(req: Request, env: Env, deps: Deps): Promise<Response> {
  let body: AnalyzeRequest;
  try { body = await req.json(); } catch { return json({ error: "bad_request" }, 400); }
  const hasImage = typeof body.image === "string" && body.image.length > 0;
  const hasText = typeof body.text === "string" && body.text.trim().length > 0;
  if ((!hasImage && !hasText) || (hasImage && body.image!.length > 2_000_000) || !["pt-PT", "en"].includes(body.locale))
    return json({ error: "bad_request" }, 400);
  if (!/^\d{2}:\d{2}$/.test(body.localTime ?? "")) body.localTime = "12:00";

  const uid = await deps.verifyRequest(req, env);
  if (!uid) return json({ error: "unauthorized" }, 401);
  if (!(await checkAndCount(env.RATE_KV, uid, dayKey(deps.now()), Number(env.DAILY_LIMIT ?? 30))))
    return json({ error: "rate_limited" }, 429);

  if (env.MOCK_AI === "1") return json(MOCK_RESULT);

  let out;
  try { out = await deps.callClaude(body, env); } catch (e) { console.error(e); return json({ error: "upstream" }, 502); }
  if (out.kind === "refusal") return json({ error: "refusal" }, 422);
  if (out.kind === "truncated") return json({ error: "bad_output" }, 422);
  try {
    const result = normalizeAiOutput(out.json);
    return result.isFood ? json(result) : json({ error: "not_food", result }, 422);
  } catch (e) {
    if (e instanceof NormalizeError) return json({ error: "bad_output" }, 422);
    throw e;
  }
}
```
